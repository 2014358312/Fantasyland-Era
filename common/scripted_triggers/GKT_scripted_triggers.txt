GAK_unit_leader_var_trigger = {
	has_country_flag = GAK_sp_griffin_commanding_tactical_doll
	OR = {
		AND = {
			check_variable = {
				GAK_unlocked_number_var < 22
			}
			is_special_project_completed = sp:sp_griffin_digi_mind_update
		}
		check_variable = {
			GAK_unlocked_number_var < 21
		}
	}
}

GAK_deploy_backup_doll_trigger = {
	any_character = {
		is_general_captured = yes
		has_trait = Command_doll
	}
	has_equipment = {
		doll_equipment_command_doll > GKT_num_command_doll
	}
}
