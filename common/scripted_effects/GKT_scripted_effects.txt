GAK_deploy_backup_doll_effect = {
	every_character = {
		limit = {
			has_trait = Command_doll
			is_general_captured = yes
		}
		release_from_captivity = yes
	}
	add_equipment_to_stockpile = {
		type = doll_equipment_command_doll
		amount = GKT_num_command_doll_equipment
	}
	set_variable = { GKT_num_command_doll = -1 }
	set_variable = { GKT_num_command_doll_equipment = 0 }
}

GKT_num_griffin_unit_effect = {
	if = {
		limit = {
			FROM = {
				check_variable = { modifier@griffin_battalions_capacity > 0 }
			}
		}	
		FROM = {
			set_variable = {
				GAK_num_griffin_battalions = {
					value = 0
					add = num_battalions_with_type@tactical_doll_ar
					add = num_battalions_with_type@tactical_doll_rf
					add = num_battalions_with_type@tactical_doll_sg
					add = num_battalions_with_type@tactical_doll_hg
					add = num_battalions_with_type@tactical_doll_smg
					add = num_battalions_with_type@tactical_doll_mg
					add = num_battalions_with_type@tactical_doll_amphibious
					add = num_battalions_with_type@tactical_doll_parachuted
					add = num_battalions_with_type@tactical_doll_third_generation
					add = num_battalions_with_type@anti_tank_unit
					add = num_battalions_with_type@auto_grenade_launcher
					add = num_battalions_with_type@mortar_unit
					add = num_battalions_with_type@main_battle_tanks_unit
					add = num_battalions_with_type@tactical_mecha_unit
				}
			}
			set_variable = {
				GAK_num_griffin_battalions_buff_trigger = {
					value = modifier@griffin_battalions_capacity
					subtract = GAK_num_griffin_battalions
					divide = modifier@griffin_battalions_capacity
					clamp = { min = -0.5 max = 0.25 }
				}
			}
			GKT_num_griffin_unit_effect_final = yes
		}
	}
	else = {
		FROM = {set_variable = { GAK_num_griffin_battalions_buff = -0.1 }}
	} 
}

GKT_num_griffin_unit_effect_final = {
	if = {
		limit = {
			FROM = {
				check_variable = { GAK_num_griffin_battalions_buff_trigger > 0 }
			}
		}
		FROM = {
			set_variable = {
				GAK_num_griffin_battalions_buff = GAK_num_griffin_battalions_buff_trigger
			}
		}
	}
	else = {
		FROM = {
			set_variable = {
				GAK_num_griffin_battalions_buff = {
					value = GAK_num_griffin_battalions_buff_trigger
					divide = -0.5
					multiply = -0.1
				}
			}
		}
	}
}
